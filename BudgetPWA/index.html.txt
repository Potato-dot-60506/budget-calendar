  <script>
    const TOTAL_BUDGET = 75 * 7 * 2;
    let weekOffset = 0;

    function formatDate(date) {
      const y = date.getFullYear();
      const m = String(date.getMonth() + 1).padStart(2, "0");
      const d = String(date.getDate()).padStart(2, "0");
      return `${y}-${m}-${d}`;
    }

    function displayDate(date) {
      return `${date.getMonth() + 1}/${date.getDate()}`;
    }

    function getTuesdayStart(baseDate) {
      const date = new Date(baseDate);
      date.setHours(0, 0, 0, 0);

      const day = date.getDay();
      const diff = day >= 2 ? day - 2 : day + 5;
      date.setDate(date.getDate() - diff);
      return date;
    }

    function addDays(date, days) {
      const d = new Date(date);
      d.setDate(d.getDate() + days);
      return d;
    }

    function getCurrentPeriodStart() {
      const today = new Date();
      const start = getTuesdayStart(today);
      start.setDate(start.getDate() + weekOffset * 7);
      return start;
    }

    function getExpenses() {
      return JSON.parse(localStorage.getItem("budgetExpenses") || "{}");
    }

    function saveExpenses(expenses) {
      localStorage.setItem("budgetExpenses", JSON.stringify(expenses));
    }

    function yuan(num) {
      return "¥" + Number(num).toFixed(2).replace(/\.00$/, "");
    }

    function getRemainingDays(periodStart) {
      const now = new Date();
      const today = new Date(now);
      today.setHours(0, 0, 0, 0);

      const periodEnd = addDays(periodStart, 6);
      periodEnd.setHours(23, 59, 59, 999);

      if (now > periodEnd) return 0;
      if (today < periodStart) return 7;

      const diff = Math.floor((periodEnd - today) / (1000 * 60 * 60 * 24)) + 1;
      return Math.max(diff, 0);
    }

    function render() {
      const calendar = document.getElementById("calendar");
      calendar.innerHTML = "";

      const start = getCurrentPeriodStart();
      const end = addDays(start, 6);
      const expenses = getExpenses();

      document.getElementById("periodText").textContent =
        `${formatDate(start)} 至 ${formatDate(end)}`;

      const weekdays = ["二", "三", "四", "五", "六", "日", "一"];
      weekdays.forEach(w => {
        const div = document.createElement("div");
        div.className = "weekday";
        div.textContent = w;
        calendar.appendChild(div);
      });

      let spent = 0;
      const todayKey = formatDate(new Date());

      for (let i = 0; i < 7; i++) {
        const date = addDays(start, i);
        const key = formatDate(date);
        const amount = Number(expenses[key] || 0);
        spent += amount;

        const btn = document.createElement("button");
        btn.className = "day" + (key === todayKey ? " today" : "");
        btn.innerHTML = `
          <strong>${displayDate(date)}</strong>
          <span class="amount">${amount ? yuan(amount) : "未记录"}</span>
        `;
        btn.onclick = () => editExpense(key, amount);
        calendar.appendChild(btn);
      }

      const remaining = TOTAL_BUDGET - spent;
      const remainingDays = getRemainingDays(start);
      const dailyPersonal =
        remainingDays > 0 ? remaining / (2 * remainingDays) : 0;

      document.getElementById("totalBudget").textContent = yuan(TOTAL_BUDGET);
      document.getElementById("spentAmount").textContent = yuan(spent);
      document.getElementById("remainingAmount").textContent = yuan(remaining);
      document.getElementById("remainingDays").textContent = remainingDays;
      document.getElementById("dailyPersonalBudget").textContent =
        yuan(dailyPersonal);
    }

    function editExpense(dateKey, currentAmount) {
      const input = prompt(`请输入 ${dateKey} 的花费金额`, currentAmount || "");
      if (input === null) return;

      const amount = Number(input);
      if (isNaN(amount) || amount < 0) {
        alert("请输入有效的非负数字");
        return;
      }

      const expenses = getExpenses();

      if (amount === 0) {
        delete expenses[dateKey];
      } else {
        expenses[dateKey] = amount;
      }

      saveExpenses(expenses);
      render();
    }

    function changeWeek(step) {
      weekOffset += step;
      render();
    }

    function goCurrentWeek() {
      weekOffset = 0;
      render();
    }

    if ("serviceWorker" in navigator) {
      navigator.serviceWorker.register("service-worker.js");
    }

    render();
  </script>
